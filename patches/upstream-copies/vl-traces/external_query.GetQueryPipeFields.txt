func getNeededColumns(pipes []pipe) *prefixfilter.Filter {
	var pf prefixfilter.Filter
	pf.AddAllowFilter("*")

	for i := len(pipes) - 1; i >= 0; i-- {
		pipes[i].updateNeededFields(&pf)
	}

	return &pf
}
